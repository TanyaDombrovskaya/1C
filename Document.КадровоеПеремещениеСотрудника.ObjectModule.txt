Процедура ОбработкаПроведения(Отказ, Режим)
	//запись в РегистрСведений.КадровыеПеремещения
	ПоследнееМесто = ПоискПоследнегоМестаРаботы(Пользователь);
	
	Если ПоследнееМесто <> Неопределено Тогда	
		ПоследнееПодразделение = ПоследнееМесто.Подразделение;
		ПоследняяДолжность = ПоследнееМесто.Должность;
	Иначе
		ПоследнееПодразделение = Неопределено;
		ПоследняяДолжность = Неопределено;
	КонецЕсли;
	
	Если ПоследнееПодразделение = ТекущееПодразделение И ПоследняяДолжность = ТекущаяДолжность Тогда
		Движения.КадровыеПеремещения.Записывать = Истина;
		Движение = Движения.КадровыеПеремещения.Добавить();
		Движение.Период = Дата;
		Движение.Сотрудник = Пользователь;
		Движение.ПоследнееПодразделение = НовоеПодразделение;
		Движение.ПоследняяДолжность = НоваяДолжность;
	Иначе
		Сообщить("Пользователь не работает на этой должности и в этом подразделении");
		Отказ = Истина;
	КонецЕсли;
КонецПроцедуры

&НаСервере
Функция ПоискПоследнегоМестаРаботы(Знач Пользователь)
	Запрос = Новый Запрос;
	
	Запрос.Текст = 
		"ВЫБРАТЬ
		|	КадровыеПеремещенияСрезПоследних.ПоследнееПодразделение КАК ПоследнееПодразделение,
		|	КадровыеПеремещенияСрезПоследних.ПоследняяДолжность КАК ПоследняяДолжность
		|ИЗ
		|	РегистрСведений.КадровыеПеремещения.СрезПоследних(, Сотрудник = &Пользователь) КАК КадровыеПеремещенияСрезПоследних";
		
	Запрос.УстановитьПараметр("Пользователь", Пользователь);
	
	РезультатЗапроса = Запрос.Выполнить();
	
	Если Не РезультатЗапроса.Пустой() Тогда
		Выборка = РезультатЗапроса.Выбрать();
		Выборка.Следующий();
		
		Результат = Новый Структура;
		Результат.Вставить("Подразделение", Выборка.ПоследнееПодразделение);
		Результат.Вставить("Должность", Выборка.ПоследняяДолжность);
		
		Возврат Результат;
	КонецЕсли;
	
	Возврат Неопределено;
КонецФункции
