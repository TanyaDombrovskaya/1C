Процедура ОпытРаботыДатаОкончанияПриИзменении(Элемент)
	ТекущаяСтрока = Элементы.ОпытРаботы.ТекущиеДанные;
	
	Если (ТекущаяСтрока.ДатаНачала > ТекущаяСтрока.ДатаОкончания) И ЗначениеЗаполнено(ТекущаяСтрока.ДатаНачала) Тогда
		Сообщить("Дата начала не может быть позже даты окончания!");
	Иначе
		ТекущаяСтрока.Срок = РасчетСрокаРаботы(ТекущаяСтрока.ДатаНачала, ТекущаяСтрока.ДатаОкончания);
	КонецЕсли;

&НаСервереБезКонтекста
Функция РасчетСрокаРаботы(Знач ДатаНачала, Знач ДатаОкончания)
	Если Не ЗначениеЗаполнено(ДатаНачала) Или Не ЗначениеЗаполнено(ДатаОкончания) Тогда
		Возврат "";
	КонецЕсли;
	
	РезультатРасчета = РассчитатьКоличествоЛетИМесяцев(ДатаНачала, ДатаОкончания);
	
	КоличествоЛет = РезультатРасчета.Лет;
	КоличествоМесяцев = РезультатРасчета.Месяцев;
	
	СклонениеЛет = СформироватьСклонениеЛет(КоличествоЛет);
	СклонениеМесяцев = СформироватьСклонениеМесяцев(КоличествоМесяцев);
	
	Результат = "" + КоличествоЛет + " " + СклонениеЛет + " " + КоличествоМесяцев + " " + СклонениеМесяцев;
	
	Возврат Результат;
КонецФункции

&НаСервереБезКонтекста
Функция РассчитатьКоличествоЛетИМесяцев(Знач ДатаНачала, Знач ДатаОкончания)
	КоличествоЛет = Год(ДатаОкончания) - Год(ДатаНачала);
	КоличествоМесяцев = Месяц(ДатаОкончания) - Месяц(ДатаНачала);
	
	Если День(ДатаОкончания) < День(ДатаНачала) Тогда
		КоличествоМесяцев = КоличествоМесяцев - 1;
	КонецЕсли;
	
	Если КоличествоМесяцев < 0 Тогда
		КоличествоЛет = КоличествоЛет - 1;
		КоличествоМесяцев = КоличествоМесяцев + 12;
	КонецЕсли;
	
	Результат = Новый Структура;
	Результат.Вставить("Лет", КоличествоЛет);
	Результат.Вставить("Месяцев", КоличествоМесяцев);
	
	Возврат Результат;
КонецФункции

&НаСервереБезКонтекста
Функция СформироватьСклонениеЛет(Знач Число)
	Склонение = "";
	
	Если (Число > 1) и (Число < 5) Тогда
		Склонение = "года";
	ИначеЕсли (Число = 1) Тогда
		Склонение = "Год";
	ИначеЕсли (Число > 4) Тогда
		Склонение = "лет";
	КонецЕсли;
	
	Возврат Склонение;
КонецФункции

&НаСервереБезКонтекста

	Склонение = "";

		Склонение = "месяца";
	ИначеЕсли (Число = 1) Тогда

	ИначеЕсли (Число > 4) Тогда
		Склонение = "месяцев";

	Возврат Склонение;
